\subsection{Simulation Setting}\label{simulation-setting}

Data Generative Model

\begin{align*}
E(Y_t|H_t, A_t) &= \{0.2 \times 1_{(Z_t = 0)} + 0.5 \times 1_{(Z_t = 1)} + 0.4 \times 1_{(Z_t = 2)}\} e^{((0.1 \times 1_{(A_t = 1)} + 0.25 \times 1_{(A_t = 1)} ) \times 0.3 Z_t)}\\
Z_t &= (0, 1, 2) \text{ with equal probability.}\\
Y_t &\sim Bern(E(Y_t | H_t, A_t))
\end{align*}
